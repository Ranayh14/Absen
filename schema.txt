CREATE TABLE `attendance` (
  `id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `jam_masuk` varchar(20) DEFAULT NULL,
  `jam_masuk_iso` datetime DEFAULT NULL,
  `ekspresi_masuk` varchar(50) DEFAULT NULL,
  `jam_pulang` varchar(20) DEFAULT NULL,
  `jam_pulang_iso` datetime DEFAULT NULL,
  `ekspresi_pulang` varchar(50) DEFAULT NULL,
  `status` enum('ontime','terlambat') DEFAULT 'ontime',
  `ket` enum('hadir','izin','sakit','alpha','wfh') DEFAULT 'hadir',
  `daily_report_id` int(11) DEFAULT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp()
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `daily_reports` (
  `id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `report_date` date NOT NULL,
  `content` text DEFAULT NULL,
  `status` enum('pending','approved','disapproved') DEFAULT 'pending',
  `evaluation` text DEFAULT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `monthly_reports` (
  `id` int(11) NOT NULL,
  `user_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `month` int(11) NOT NULL,
  `summary` text DEFAULT NULL,
  `achievements` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL CHECK (json_valid(`achievements`)),
  `obstacles` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL CHECK (json_valid(`obstacles`)),
  `status` enum('draft','submitted','approved','disapproved') DEFAULT 'draft',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `users` (
  `id` int(11) NOT NULL,
  `role` enum('admin','pegawai') NOT NULL DEFAULT 'pegawai',
  `email` varchar(255) NOT NULL,
  `nim` varchar(100) DEFAULT NULL,
  `nama` varchar(255) NOT NULL,
  `prodi` varchar(255) DEFAULT NULL,
  `startup` varchar(255) DEFAULT NULL,
  `foto_base64` longtext DEFAULT NULL,
  `password_hash` varchar(255) NOT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp()
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
