CREATE TABLE `admin_help_requests` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `request_type` enum('past_attendance','late_attendance','bug_report') NOT NULL,
  `attendance_type` enum('wfo','wfa','overtime') DEFAULT 'wfo',
  `attendance_reason` text DEFAULT NULL,
  `tanggal` date DEFAULT NULL,
  `jam_masuk` time DEFAULT NULL,
  `jam_pulang` time DEFAULT NULL,
  `alasan_izin` text DEFAULT NULL,
  `jenis_izin` enum('izin','sakit') DEFAULT NULL,
  `bukti_izin` longtext DEFAULT NULL,
  `bukti_presensi` longtext DEFAULT NULL,
  `lokasi_presensi` varchar(255) DEFAULT NULL,
  `bug_description` text DEFAULT NULL,
  `bug_proof` longtext DEFAULT NULL,
  `status` enum('pending','approved','disapproved','solved') DEFAULT 'pending',
  `admin_note` text DEFAULT NULL,
  `is_read_by_user` tinyint(1) DEFAULT 0,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`),
  KEY `user_id` (`user_id`),

  `jam_masuk_iso` datetime DEFAULT NULL,

  `lng_masuk` decimal(10,7) DEFAULT NULL,
  `jam_pulang` varchar(20) DEFAULT NULL,
  `jam_pulang_iso` datetime DEFAULT NULL,
  `ekspresi_pulang` varchar(50) DEFAULT NULL,
  `screenshot_pulang` longtext DEFAULT NULL,
  `lokasi_pulang` varchar(255) DEFAULT NULL,
  `lat_pulang` decimal(10,7) DEFAULT NULL,
  `lng_pulang` decimal(10,7) DEFAULT NULL,
  `status` enum('ontime','terlambat','-') DEFAULT 'ontime',
  `ket` enum('wfo','izin','sakit','alpha','wfa','overtime') DEFAULT 'wfo',
  `alasan_wfa` text DEFAULT NULL,

CREATE TABLE `daily_reports` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `report_date` date NOT NULL,
  `content` text DEFAULT NULL,
  `status` enum('pending','approved','disapproved') DEFAULT 'pending',
  `evaluation` text DEFAULT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_user_date` (`user_id`,`report_date`),
  CONSTRAINT `fk_dr_user` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=1494 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `monthly_reports` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `year` int(11) NOT NULL,
  `month` int(11) NOT NULL,
  `summary` text DEFAULT NULL,
  `achievements` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL,
  `obstacles` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL,
  `status` enum('draft','belum di approve','approved','disapproved') DEFAULT 'draft',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_user_month` (`user_id`,`year`,`month`),
  CONSTRAINT `fk_mr_user` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE,
  CONSTRAINT `monthly_reports_chk_1` CHECK (json_valid(`achievements`)),
  CONSTRAINT `monthly_reports_chk_2` CHECK (json_valid(`obstacles`))
) ENGINE=InnoDB AUTO_INCREMENT=23 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
